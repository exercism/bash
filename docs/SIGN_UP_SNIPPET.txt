most_common_word() {
    local -A counts
    local word best="" best_count=0
    for word in "${@,,}"; do
        (( counts[$word]++ ))
        if (( counts[$word] > best_count )); then
            best="${word}"
            best_count="${counts[$word]}"
        fi
    done
    printf "%s appears %d times\n" "${best@Q}" "${best_count}"
}

most_common_word "$@"
